# Auto-Smaart Ring Out Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the FIXED-only Ring Out mode that raises one Quantum 225 strip fader until sustained RTA peaks appear, auto-writes up to eight −3.0 dB Q 16.00 notches, stops at +5.0 dB above first feedback or at eight bands, and shares mute-all Stop with Calibration.

**Architecture:** Pure peak detection and notch table rules live in `AutoSmaartCore`. The existing `SmaartAPI` session gains an RTA/spectrum read. The existing `ConsoleOSC` client gains strip fader, strip EQ, and mute-all addresses. An `AutoSmaartRingOut` controller owns Start, the raise/notch loop, undo snapshot, and Stop cancel. SwiftUI draws one Ring Out page and sends nothing by itself. Controller phases stay internal. They do not switch the page.

**Tech Stack:** Swift 5.10 package already on `cursor/auto-calibration-impl-d933`, macOS 14, SwiftUI, Foundation URLSession WebSockets, POSIX UDP OSC, XCTest. Run every `swift test` command on a Mac.

```swift
import Foundation

public enum StripKind: String, Equatable, Codable, CaseIterable {
    case inputChannel
    case groupBus
    case auxBus
    case matrixBus
}

public func stripIndexRange(_ kind: StripKind) -> ClosedRange<Int> {
    switch kind {
    case .inputChannel: return 1...72
    case .groupBus, .auxBus: return 1...24
    case .matrixBus: return 1...12
    }
}

public enum FaderLevel: Equatable {
    case minusInfinity
    case decibels(Double)
}

public let digicoFaderMinusInfinityWire: Float = -150.0

public func faderLevel(fromWire value: Float) -> FaderLevel {
    if value <= digicoFaderMinusInfinityWire { return .minusInfinity }
    return .decibels(Double(value))
}

public func faderWire(from level: FaderLevel) -> Float {
    switch level {
    case .minusInfinity: return digicoFaderMinusInfinityWire
    case .decibels(let db): return Float(db)
    }
}

public func raisedFader(_ level: FaderLevel, by stepDb: Double) -> FaderLevel {
    switch level {
    case .minusInfinity: return .decibels(Double(digicoFaderMinusInfinityWire) + stepDb)
    case .decibels(let db): return .decibels(db + stepDb)
    }
}

public func faderDbComparable(_ level: FaderLevel) -> Double {
    switch level {
    case .minusInfinity: return -Double.infinity
    case .decibels(let db): return db
    }
}
```

```swift
import Foundation

public struct PeakFrame: Equatable {
    public var binIndex: Int
    public var hz: Double
    public var magnitudeDb: Double
    public init(binIndex: Int, hz: Double, magnitudeDb: Double) {
        self.binIndex = binIndex
        self.hz = hz
        self.magnitudeDb = magnitudeDb
    }
}

public struct SustainedPeak: Equatable {
    public var binIndex: Int
    public var hz: Double
    public var magnitudeDb: Double
    public init(binIndex: Int, hz: Double, magnitudeDb: Double) {
        self.binIndex = binIndex
        self.hz = hz
        self.magnitudeDb = magnitudeDb
    }
}

public func peakFrame(in spectrum: [MagnitudePoint]) -> PeakFrame? {
    let kept = spectrum.enumerated().filter { $0.element.hz >= 100 && $0.element.hz <= 8000 }
    guard !kept.isEmpty else { return nil }
    let magnitudes = kept.map(\.element.db)
    let sorted = magnitudes.sorted()
    let median: Double
    if sorted.count % 2 == 0 {
        median = (sorted[sorted.count / 2 - 1] + sorted[sorted.count / 2]) / 2.0
    } else {
        median = sorted[sorted.count / 2]
    }
    var best = 0
    for index in kept.indices.dropFirst() {
        let left = kept[best].element
        let right = kept[index].element
        if right.db > left.db || (right.db == left.db && right.hz < left.hz) {
            best = index
        }
    }
    let candidate = kept[best]
    guard candidate.element.db >= median + 10.0 else { return nil }
    var low = best
    while low > 0 && abs(kept[low - 1].element.db - candidate.element.db) <= 3.0 { low -= 1 }
    var high = best
    while high < kept.count - 1 && abs(kept[high + 1].element.db - candidate.element.db) <= 3.0 { high += 1 }
    let fLo = kept[low].element.hz
    let fHi = kept[high].element.hz
    let narrow: Bool
    if fLo == fHi {
        narrow = true
    } else {
        let n = log2(fHi / fLo)
        let q = sqrt(pow(2.0, n)) / (pow(2.0, n) - 1.0)
        narrow = q >= 10.0
    }
    guard narrow else { return nil }
    return PeakFrame(binIndex: candidate.offset, hz: candidate.element.hz, magnitudeDb: candidate.element.db)
}

public final class SustainedPeakDetector {
    private let requiredFrames: Int
    private var streakBin: Int?
    private var magnitudes: [Double] = []

    public init(requiredFrames: Int = 8) {
        self.requiredFrames = requiredFrames
    }

    public func reset() {
        streakBin = nil
        magnitudes = []
    }

    public func push(_ spectrum: [MagnitudePoint]) -> SustainedPeak? {
        guard let frame = peakFrame(in: spectrum) else {
            reset()
            return nil
        }
        if streakBin != frame.binIndex {
            streakBin = frame.binIndex
            magnitudes = [frame.magnitudeDb]
        } else {
            magnitudes.append(frame.magnitudeDb)
        }
        guard magnitudes.count >= requiredFrames, let bin = streakBin else { return nil }
        let mean = magnitudes.reduce(0, +) / Double(magnitudes.count)
        let peak = SustainedPeak(binIndex: bin, hz: frame.hz, magnitudeDb: mean)
        reset()
        return peak
    }
}
```

**Interfaces:**
- Consumes: `EqBand`

  - `public func centsDistance(_ aHz: Double, _ bHz: Double) -> Double` — `1200 * abs(log2(a/b))`

Match when cents ≤ 50. Closest cents wins; ties → lower `index`. New band uses lowest free index 1…8, gain −3.0, Q 16.00. Deepen adds −3.0 to gain; if already at −18.0 or would go below, `.floorReached` with gain left at −18.0. Unmatched with 8 bands → `.bandCap`.

```swift
import Foundation

public enum NotchDecision: Equatable {
    case write(EqBand)
    case deepen(EqBand)
    case bandCap
    case floorReached(EqBand)
}

public func centsDistance(_ aHz: Double, _ bHz: Double) -> Double {
    1200.0 * abs(log2(aHz / bHz))
}

public func decideNotch(peakHz: Double, bands: [EqBand]) -> NotchDecision {
    let matches = bands
        .map { (band: $0, cents: centsDistance(peakHz, $0.hz)) }
        .filter { $0.cents <= 50 }
        .sorted { lhs, rhs in
            if lhs.cents == rhs.cents { return lhs.band.index < rhs.band.index }
            return lhs.cents < rhs.cents
        }
    if let match = matches.first?.band {
        if match.gainDb <= -18.0 {
            return .floorReached(match)
        }
        let nextGain = max(-18.0, match.gainDb - 3.0)
        return .deepen(EqBand(index: match.index, hz: match.hz, gainDb: nextGain, q: match.q))
    }
    let used = Set(bands.map(\.index))
    guard let free = (1...8).first(where: { !used.contains($0) }) else {
        return .bandCap
    }
    return .write(EqBand(index: free, hz: peakHz, gainDb: -3.0, q: 16.0))
}
```

```swift
public enum AddressError: Error, Equatable {
    case outputOutOfRange
    case bandOutOfRange
    case stripOutOfRange

public func stripPrefix(kind: StripKind, index: Int) throws -> String {
    guard stripIndexRange(kind).contains(index) else { throw AddressError.stripOutOfRange }
    switch kind {
    case .inputChannel: return "/sd/Input_Channels/\(index)"
    case .groupBus: return "/sd/Group_Outputs/\(index)"
    case .auxBus: return "/sd/Aux_Outputs/\(index)"
    case .matrixBus: return "/sd/Matrix_Outputs/\(index)"
    }
}

public func faderAddress(kind: StripKind, index: Int) throws -> String {
    try stripPrefix(kind: kind, index: index) + "/fader"
}

public func eqAddresses(kind: StripKind, index: Int, band: Int) throws -> (freq: String, gain: String, q: String, on: String) {
    guard (1...8).contains(band) else { throw AddressError.bandOutOfRange }
    let prefix = try stripPrefix(kind: kind, index: index) + "/EQ"
    return (
        prefix + "/eq_freq_\(band)",
        prefix + "/eq_gain_\(band)",
        prefix + "/eq_Q_\(band)",
        prefix + "/eq_on_\(band)"
    )
}

public func delayAddresses(output: Int) throws -> (delay: String, fineDelay: String) {
    guard (1...12).contains(output) else { throw AddressError.outputOutOfRange }
```

```swift
import Foundation
import AutoSmaartCore
import ConsoleOSC

public final class NotchWriter {
    private let kind: StripKind
    private let index: Int
    private let link: OscLink
    public var bands: [EqBand] = []
    public private(set) var undoSnapshot: [Replacement] = []

    public init(kind: StripKind, index: Int, link: OscLink) {
        self.kind = kind
        self.index = index
        self.link = link
    }

    public func captureUndoBaseline(readingBands bandIndexes: [Int]) throws {
        var snapshot: [Replacement] = []
        for band in bandIndexes {
            let eq = try eqAddresses(kind: kind, index: index, band: band)
            for address in [eq.freq, eq.gain, eq.q, eq.on] {
                do {
                    snapshot.append(Replacement(address: address, value: try link.read(address: address)))
                } catch {
                    // Unused band with no console read: skip this address.
                }
            }
        }
        undoSnapshot = snapshot
    }

    public func apply(peakHz: Double) throws -> NotchDecision {
        let decision = decideNotch(peakHz: peakHz, bands: bands)
        switch decision {
        case .bandCap, .floorReached:
            return decision
        case .write(let band):
            let eq = try eqAddresses(kind: kind, index: index, band: band.index)
            let items = [
                Replacement(address: eq.freq, value: .float32(Float(band.hz))),
                Replacement(address: eq.gain, value: .float32(Float(band.gainDb))),
                Replacement(address: eq.q, value: .float32(Float(band.q))),
                Replacement(address: eq.on, value: .int32(1))
            ]
            try writeOnly(items, link: link)
            bands.append(band)
            bands.sort { $0.index < $1.index }
            return decision
        case .deepen(let band):
            let eq = try eqAddresses(kind: kind, index: index, band: band.index)
            let items = [
                Replacement(address: eq.freq, value: .float32(Float(band.hz))),
                Replacement(address: eq.gain, value: .float32(Float(band.gainDb))),
                Replacement(address: eq.q, value: .float32(Float(band.q))),
                Replacement(address: eq.on, value: .int32(1))
            ]
            _ = try confirmReplace(items, link: link)
            if let offset = bands.firstIndex(where: { $0.index == band.index }) {
                bands[offset] = band
            }
            return decision
        }
    }

    public func undo() throws {
        guard !undoSnapshot.isEmpty else {
            bands = []
            return
        }
        try writeOnly(undoSnapshot, link: link)
        bands = []
    }
}
```

New-band path uses `writeOnly` (four writes, no reads). Deepen path uses `confirmReplace`.

```swift
    private let linkFactory: (String, UInt16) throws -> OscLink
    private var link: OscLink?
    private var fader: FaderDriver?
    private var notches: NotchWriter?
    private var detector = SustainedPeakDetector()
    private var lastRaise: Date?
    private var rtaMissingSince: Date?
    private var active = false
    @Published public private(set) var state = RingOutState()

    public init(smaart: SmaartReader, linkFactory: @escaping (String, UInt16) throws -> OscLink) {
        self.smaart = smaart
        self.linkFactory = linkFactory
    }

    public func updateHost(_ host: String) {
        guard !state.loopLocked else { return }
        state.host = host
    }

    public func updatePort(_ port: String) {
        guard !state.loopLocked else { return }
        state.port = port
    }

    public func selectFader(kind: StripKind, index: Int) {
        guard !state.loopLocked else { return }
        guard stripIndexRange(kind).contains(index) else { return }
        state.faderKind = kind
        state.faderIndex = index
    }

    public func selectEQ(kind: StripKind, index: Int) {
        guard !state.loopLocked else { return }
        guard stripIndexRange(kind).contains(index) else { return }
        state.eqKind = kind
        state.eqIndex = index
    }

    public func start() {
        guard !active else { return }
        state.notice = .none
        if state.host.isEmpty || state.port.isEmpty {
            state.notice = .hostOrPortEmpty
            return
        }
        guard let port = UInt16(state.port) else {
            state.notice = .hostOrPortEmpty
            return
        }
        do {
            guard try smaart.hasActiveRTA() else {
                state.notice = .noRTA
                return
            }
            let link = try linkFactory(state.host, port)
            self.link = link
            let fader = FaderDriver(kind: state.faderKind, index: state.faderIndex, link: link)
            let notches = NotchWriter(kind: state.eqKind, index: state.eqIndex, link: link)
            let startLevel = try fader.read()
            try notches.captureUndoBaseline(readingBands: Array(1...8))
            self.fader = fader
            self.notches = notches
            detector.reset()
            lastRaise = nil
            rtaMissingSince = nil
            state.startFader = startLevel
            state.currentFader = startLevel
            state.firstFeedbackFader = nil
            state.stopFader = nil
            state.bands = []
            state.successfulNotchWrites = 0
            state.undoAvailable = false
            state.phase = .running
            state.loopLocked = true
            state.rtaLive = true
            active = true
            publishBands()
        } catch {
            state.notice = .noRTA
            active = false
            state.loopLocked = false
            state.phase = .setup
        }
    }

    public func tick(now: Date) {
        guard active else {
            refreshTrace()
            return
        }
        guard state.phase == .running else { return }
        guard let fader, let notches, link != nil else { return }

        let rta: [MagnitudePoint]
        do {
            rta = try smaart.readRTA()
            state.latestRTA = rta
            state.rtaLive = true
            rtaMissingSince = nil
        } catch {
            state.rtaLive = false
            state.latestRTA = []
            if rtaMissingSince == nil { rtaMissingSince = now }
            if let missing = rtaMissingSince, now.timeIntervalSince(missing) >= 2.0 {
                finishIdle(notice: .noRTA)
            }
            return
        }

        if lastRaise == nil || now.timeIntervalSince(lastRaise!) >= 0.2 {
            do {
                let level = try fader.raiseByHalfDb()
                state.currentFader = level
                lastRaise = now
                if let first = state.firstFeedbackFader {
                    let stopDb = faderDbComparable(first) + 5.0
                    if faderDbComparable(level) >= stopDb {
                        state.stopFader = .decibels(stopDb)
                        state.phase = .done
                        state.notice = .completePlus5
                        active = false
                        state.loopLocked = false
                        state.undoAvailable = state.successfulNotchWrites > 0
                        publishBands()
                        return
                    }
                }
            } catch {
                state.notice = .faderWriteFailed
                finishIdle(notice: .faderWriteFailed)
                return
            }
        }

        if let peak = detector.push(rta) {
            handlePeak(peakHz: peak.hz, fader: fader, notches: notches)
        }
        publishBands()
    }

    private func handlePeak(peakHz: Double, fader: FaderDriver, notches: NotchWriter) {
        state.phase = .notch
        if state.firstFeedbackFader == nil {
            let f0 = state.currentFader ?? .minusInfinity
            state.firstFeedbackFader = f0
            state.stopFader = .decibels(faderDbComparable(f0) + 5.0)
        }
        do {
            try fader.writeMinusInfinity()
            state.currentFader = .minusInfinity
        } catch {
            finishIdle(notice: .faderWriteFailed)
            return
        }
        do {
            let decision = try notches.apply(peakHz: peakHz)
            switch decision {
            case .write, .deepen:
                state.successfulNotchWrites += 1
                state.phase = .running
                detector.reset()
                lastRaise = nil
            case .bandCap:
                state.phase = .done
                state.notice = .eightBands
                active = false
                state.loopLocked = false
                state.undoAvailable = state.successfulNotchWrites > 0
            case .floorReached:
                state.phase = .done
                state.notice = .notchFloor
                active = false
                state.loopLocked = false
                state.undoAvailable = state.successfulNotchWrites > 0
            }
            publishBands()
        } catch let error as ConfirmError {
            state.currentFader = .minusInfinity
            switch error {
            case .kept: finishIdle(notice: .notchWriteFailedRestored)
            case .mismatch: finishIdle(notice: .notchWriteFailedMismatch)
            }
        } catch {
            finishIdle(notice: .notchWriteFailedRestored)
        }
    }

    public func stop() {
        active = false
        state.loopLocked = false
        state.phase = .setup
        if let link {
            let result = muteAll(link: link)
            state.notice = result.accepted == result.attempted ? .stopMuted : .stopMuteIncomplete
        } else {
            state.notice = .stopMuted
        }
        state.undoAvailable = state.successfulNotchWrites > 0
        publishBands()
    }

    public func undoNotches() {
        guard !active else { return }
        guard state.undoAvailable, let notches else { return }
        do {
            try notches.undo()
            state.bands = []
            state.successfulNotchWrites = 0
            state.undoAvailable = false
            state.notice = .none
        } catch {
            state.notice = .notchWriteFailedMismatch
        }
    }

    private func finishIdle(notice: RingOutNotice) {
        active = false
        state.loopLocked = false
        state.phase = .setup
        state.notice = notice
        state.undoAvailable = state.successfulNotchWrites > 0
        publishBands()
    }

    private func refreshTrace() {
        do {
            guard try smaart.hasActiveRTA() else {
                state.rtaLive = false
                state.latestRTA = []
                return
            }
            state.latestRTA = try smaart.readRTA()
            state.rtaLive = true
        } catch {
            state.rtaLive = false
            state.latestRTA = []
        }
    }

    private func publishBands() {
        state.bands = notches?.bands ?? state.bands
    }
}
```

```swift
import Foundation
import AutoSmaartCore

public enum RingOutPresenter {
    public static let smaartLive = "Smaart live"
    public static let startAction = "Start ring-out"
    public static let undoAction = "Undo notches"
    public static let stopAction = "Stop"
    public static let leaveAction = "Calibration"
    public static let faderTargetHeading = "Fader"
    public static let eqTargetHeading = "EQ write"
    public static let modeLockNote = "Mode switch locked while loop runs. Stop first."
    public static let emDash = "—"

    public static func noticeText(_ notice: RingOutNotice) -> String {
        switch notice {
        case .none: return ""
        case .noRTA: return "No RTA"
        case .stopMuted: return "System muted. Loop idle."
        case .stopMuteIncomplete: return "Stop: mute-all incomplete. Check the console."
        case .notchWriteFailedRestored: return "Notch write failed. Previous EQ restored."
        case .notchWriteFailedMismatch: return "Notch write failed. Console EQ may not match the previous value."
        case .faderWriteFailed: return "Fader write failed."
        case .eightBands: return "8 bands used before +5 dB headroom."
        case .notchFloor: return "Notch floor reached."
        case .completePlus5: return "Ring-out complete (+5 dB)."
        case .hostOrPortEmpty: return "Enter the console host and OSC port."
        }
    }

    public static func statusText(rtaLive: Bool, notice: RingOutNotice, host: String, port: String) -> String {
        if notice != .none { return noticeText(notice) }
        if host.isEmpty || port.isEmpty { return noticeText(.hostOrPortEmpty) }
        return rtaLive ? smaartLive : noticeText(.noRTA)
    }

    public static func kindName(_ kind: StripKind) -> String {
        switch kind {
        case .inputChannel: return "Input"
        case .groupBus: return "Group"
        case .auxBus: return "Aux"
        case .matrixBus: return "Matrix"
        }
    }

    public static func stripLabel(kind: StripKind, index: Int) -> String {
        switch kind {
        case .inputChannel: return "Ch \(index)"
        case .groupBus: return "Group \(index)"
        case .auxBus: return "Aux \(index)"
        case .matrixBus: return "Matrix \(index)"
        }
    }

    public static func targetLabel(faderKind: StripKind, faderIndex: Int, eqKind: StripKind, eqIndex: Int) -> String {
        let fader = stripLabel(kind: faderKind, index: faderIndex)
        if faderKind == eqKind && faderIndex == eqIndex { return fader }
        return "\(fader) · EQ \(stripLabel(kind: eqKind, index: eqIndex))"
    }

    public static func frequencyCell(_ band: EqBand?) -> String {
        guard let band else { return emDash }
        return formatFrequency(band.hz)
    }

    public static func gainCell(_ band: EqBand?) -> String {
        guard let band else { return emDash }
        let rounded = roundHalfAwayFromZero(band.gainDb, places: 1)
        let sign = rounded > 0 ? "+" : (rounded < 0 ? "−" : "")
        return "\(sign)\(String(format: "%.1f", abs(rounded))) dB"
    }

    public static func qCell(_ band: EqBand?) -> String {
        guard let band else { return emDash }
        return String(format: "%.2f", band.q)
    }

    public static func faderCell(_ level: FaderLevel?) -> String {
        guard let level else { return emDash }
        switch level {
        case .minusInfinity: return "−∞"
        case .decibels(let db):
            let rounded = roundHalfAwayFromZero(db, places: 1)
            let sign = rounded > 0 ? "+" : (rounded < 0 ? "−" : "")
            return "\(sign)\(String(format: "%.1f", abs(rounded))) dB"
        }
    }

}
```

```swift
    private func faderFraction(_ level: FaderLevel?) -> CGFloat {
        guard let level else { return 0 }
        switch level {
        case .minusInfinity:
            return 0
        case .decibels(let db):
            let clamped = min(10.0, max(-60.0, db))
            return CGFloat((clamped + 60.0) / 70.0)
        }
    }

struct RTAPlot: View {
    let points: [MagnitudePoint]
    let bands: [EqBand]

    var body: some View {
        VStack(alignment: .leading, spacing: 4) {
            Canvas { context, size in
                let kept = points.filter { $0.hz >= 100 && $0.hz <= 8000 }
                var trace = Path()
                for (offset, point) in kept.enumerated() {
                    let spot = CGPoint(x: xPos(point.hz, width: size.width), y: yPos(point.db, height: size.height))
                    if offset == 0 { trace.move(to: spot) } else { trace.addLine(to: spot) }
                }
                context.stroke(trace, with: .color(Theme.live), lineWidth: 1.5)
                for band in bands where band.hz >= 100 && band.hz <= 8000 {
                    var mark = Path()
                    let x = xPos(band.hz, width: size.width)
                    mark.move(to: CGPoint(x: x, y: 0))
                    mark.addLine(to: CGPoint(x: x, y: size.height))
                    context.stroke(mark, with: .color(Theme.gold), lineWidth: 2)
                }
            }
            HStack {
                Text(formatFrequency(100))
                Spacer()
                Text(formatFrequency(8000))
            }
            .font(.system(size: 11))
        }
    }

    private func xPos(_ hz: Double, width: CGFloat) -> CGFloat {
        let lo = log10(100.0)
        let hi = log10(8000.0)
        let clamped = min(8000.0, max(100.0, hz))
        return CGFloat((log10(clamped) - lo) / (hi - lo)) * width
    }

    private func yPos(_ db: Double, height: CGFloat) -> CGFloat {
        let clamped = min(0.0, max(-80.0, db))
        return height - CGFloat((clamped + 80.0) / 80.0) * height
    }
}
```
